```ts
// XO брэнд: XO_* орчны хувьсагч үндсэн, PAPERCLIP_* нөөц (spec §10.1).
const XO_PREFIX = "XO_";
const LEGACY_PREFIX = "PAPERCLIP_";

type Env = Record<string, string | undefined>;

// XO-ийн өөрийн хувьсагч: PAPERCLIP_ хослолгүй, alias хийхгүй.
export const XO_NATIVE_ENV_KEYS: ReadonlySet<string> = new Set(["XO_I18N_TRANSFORM", "XO_SCREENSHOTS"]);

export function brandEnvSuffix(key: string): string | null {
  if (key.startsWith(LEGACY_PREFIX)) return key.slice(LEGACY_PREFIX.length) || null;
  if (key.startsWith(XO_PREFIX)) return key.slice(XO_PREFIX.length) || null;
  return null;
}

export function isBrandRuntimeEnvKey(key: string): boolean {
  return brandEnvSuffix(key) !== null;
}

export function readBrandEnv(name: string, env: Env = process.env): string | undefined {
  const xo = env[XO_PREFIX + name];
  if (xo !== undefined && xo.trim() !== "") return xo;
  return env[LEGACY_PREFIX + name];
}

// Процесс эхлэхэд: XO_X байвал PAPERCLIP_X-ийг дарж бичнэ, ингэснээр
// PAPERCLIP_* уншдаг одоогийн код XO утгыг авна.
export function applyXoEnvAliases(env: Env = process.env): void {
  for (const [key, value] of Object.entries(env)) {
    if (!key.startsWith(XO_PREFIX) || XO_NATIVE_ENV_KEYS.has(key)) continue;
    if (value === undefined || value.trim() === "") continue;
    env[LEGACY_PREFIX + key.slice(XO_PREFIX.length)] = value;
  }
}

// Агентын child env: PAPERCLIP_X бүрд XO_X хуулбар нэмнэ.
export function mirrorPaperclipEnvToXo<T extends Env>(env: T): T {
  const out: Env = { ...env };
  for (const [key, value] of Object.entries(env)) {
    if (!key.startsWith(LEGACY_PREFIX) || value === undefined) continue;
    out[XO_PREFIX + key.slice(LEGACY_PREFIX.length)] = value;
  }
  return out as T;
}
```

#!/usr/bin/env node
// XO брэнд скан: хэрэглэгчид харагдах эх дахь "Paperclip" тоог baseline-тэй
// харьцуулна. Upstream синкийн дараа шинэ тохиолдол орж ирвэл exit 1.
import fs from "node:fs";
import path from "node:path";
import { fileURLToPath } from "node:url";

const BRAND_RE = /(?<![\w-])Paperclip(?![\w-])/g;
const ROOTS = [
  { dir: "ui/src", exts: [".ts", ".tsx"] },
  { dir: "server/src/onboarding-assets", exts: [".md"] },
];
const SKIP_FILE_RE = /\.(test|spec|stories)\.[jt]sx?$/;
const SKIP_DIRS = new Set(["node_modules", "__tests__"]);

export function countBrand(text) {
  return (text.match(BRAND_RE) ?? []).length;
}

function walk(dir, exts, out) {
  for (const entry of fs.readdirSync(dir, { withFileTypes: true })) {
    const full = path.join(dir, entry.name);
    if (entry.isDirectory()) {
      if (!SKIP_DIRS.has(entry.name)) walk(full, exts, out);
    } else if (exts.includes(path.extname(entry.name)) && !SKIP_FILE_RE.test(entry.name)) {
      out.push(full);
    }
  }
}

export function scan(repoRoot) {
  const counts = [];
  for (const { dir, exts } of ROOTS) {
    const abs = path.join(repoRoot, dir);
    if (!fs.existsSync(abs)) continue;
    const files = [];
    walk(abs, exts, files);
    for (const file of files) {
      const n = countBrand(fs.readFileSync(file, "utf8"));
      if (n > 0) counts.push([path.relative(repoRoot, file).split(path.sep).join("/"), n]);
    }
  }
  counts.sort(([a], [b]) => a.localeCompare(b));
  return Object.fromEntries(counts);
}

export function compare(current, baseline) {
  const regressions = [];
  for (const [file, count] of Object.entries(current)) {
    const allowed = baseline[file] ?? 0;
    if (count > allowed) regressions.push({ file, count, allowed });
  }
  return regressions;
}

function main() {
  const repoRoot = path.resolve(path.dirname(fileURLToPath(import.meta.url)), "../..");
  const baselinePath = path.join(repoRoot, "scripts/xo-brand/baseline.json");
  const current = scan(repoRoot);
  const fileCount = Object.keys(current).length;
  const total = Object.values(current).reduce((sum, n) => sum + n, 0);

  if (process.argv.includes("--update")) {
    fs.writeFileSync(baselinePath, `${JSON.stringify(current, null, 2)}\n`);
    console.log(`xo-brand: baseline updated (${fileCount} files, ${total} hits)`);
    return;
  }
  if (process.argv.includes("--list")) {
    for (const [file, n] of Object.entries(current)) console.log(`${n}\t${file}`);
  }
  const baseline = fs.existsSync(baselinePath) ? JSON.parse(fs.readFileSync(baselinePath, "utf8")) : {};
  const regressions = compare(current, baseline);
  for (const r of regressions) console.error(`xo-brand: ${r.file}: ${r.count} > baseline ${r.allowed}`);
  if (regressions.length > 0) process.exit(1);
  console.log(`xo-brand: OK (${fileCount} files, ${total} hits, none above baseline)`);
}

if (process.argv[1] && path.resolve(process.argv[1]) === fileURLToPath(import.meta.url)) main();

// X-Paperclip-Run-Id, @paperclipai зэрэг тодорхойлогчид хүрэхгүй.
const LEGACY_BRAND_RE = /(?<![\w-])Paperclip(?![\w-])/g;

function translateCatalog(message: string): string {
  if (resolveLocale() !== "mn") return message;
  if (Object.hasOwn(catalog, message)) return catalog[message];
  for (const entry of templates) {
    const match = entry.pattern.exec(message);
    if (!match) continue;
    const values = new Map<string, string>();
    entry.names.forEach((name, index) => {
      if (!values.has(name)) values.set(name, match[index + 1]);
    });
    return entry.translation.replace(/\{\{(\w+)\}\}/g, (_, name: string) => values.get(name) ?? "");
  }
  return message;
}

export function translateServerMessage(message: string): string {
  return translateCatalog(message).replace(LEGACY_BRAND_RE, BRAND.name);
}
